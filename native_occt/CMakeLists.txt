cmake_minimum_required(VERSION 3.20)
project(OcctHmiBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCASCADE REQUIRED)
add_library(OcctHmiBridge SHARED OcctHmiBridge.cpp OcctHmiBridge.h)
target_include_directories(OcctHmiBridge PRIVATE ${OpenCASCADE_INCLUDE_DIR} ${OpenCASCADE_INCLUDE_DIRS})
if(OpenCASCADE_LIBRARIES)
  target_link_libraries(OcctHmiBridge PRIVATE ${OpenCASCADE_LIBRARIES})
else()
  target_link_libraries(OcctHmiBridge PRIVATE TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo)
endif()
set_target_properties(OcctHmiBridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(cfg Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER ${cfg} CFGU)
  set_target_properties(OcctHmiBridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CFGU} "${CMAKE_BINARY_DIR}/bin/${cfg}")
endforeach()
